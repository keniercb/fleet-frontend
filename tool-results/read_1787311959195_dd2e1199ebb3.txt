     1→import { useState, useEffect, useCallback } from 'react';
     2→import { Plus, Pencil, Trash2, Search, ChevronDown, Eye, BarChart3, ChevronRight } from 'lucide-react';
     3→import { useCrud } from '@/hooks/useCrud';
     4→import { useToast } from '@/contexts/ToastContext';

     8→  tiposVehiculoApi,
     9→  marcasApi,
    10→  tiposCombustibleApi,
    11→  choferesApi,
    12→} from '@/api/endpoints';
    13→import PageHeader from '@/components/common/PageHeader';
    14→import Pagination from '@/components/common/Pagination';
    15→import Modal from '@/components/ui/Modal';
    16→import ConfirmModal from '@/components/ui/ConfirmModal';
    17→import type {
    18→  VehiculoRequest,
    19→  VehiculoResponse,

    27→

    54→  empresaId: 0,

    58→  tipoCombustibleId: 0,
    59→  modelo: '',
    60→  matricula: '',
    61→  numeroMotor: '',
    62→  odometro: 0,
    63→  combustible: 0,
    64→  ultimoMantenimiento: '',
    65→  odometroUltimoMantenimiento: '',
    66→  indiceConsumo: '',

    68→
    69→// ---- Componentes auxiliares ----
    70→
    71→function formatDate(dateStr: string): string {
    72→  if (!dateStr) return '—';
    73→  const date = new Date(dateStr + (dateStr.length === 10 ? 'T00:00:00' : ''));
    74→  return date.toLocaleDateString('es-ES', { day: '2-digit', month: '2-digit', year: 'numeric' });
    75→}
    76→
    77→function DetailField({ label, value }: { label: string; value: string }) {
    78→  return (
    79→    <div>
    80→      <dt className="text-xs font-medium text-gray-500">{label}</dt>
    81→      <dd className="mt-0.5 text-sm text-gray-900 font-medium">{value}</dd>
    82→    </div>
    83→  );
    84→}

    86→// ---- Component ----
    87→
    88→export default function VehiculosPage() {
    89→  const {
    90→    data, loading, saving, totalPages, totalElements, page, size, error,
    91→    setPage, createItem, updateItem, deleteItem,
    92→  } = useCrud<VehiculoRequest, VehiculoResponse>(vehiculosApi);
    93→
    94→  const { addToast } = useToast();
    95→
    96→  // Dropdown data
    97→  const [dropdowns, setDropdowns] = useState<Dropdowns>({
    98→    empresas: [],
    99→    tiposVehiculo: [],
   100→    marcas: [],
   101→    tiposCombustible: [],
   102→  });
   103→
   104→  // Choferes filtrados por empresa
   105→  const [choferes, setChoferes] = useState<ChoferResponse[]>([]);
   106→  const [loadingChoferes, setLoadingChoferes] = useState(false);
   107→
   108→  // UI state
   109→  const [search, setSearch] = useState('');
   110→  const [showForm, setShowForm] = useState(false);
   111→  const [viewEntity, setViewEntity] = useState<VehiculoResponse | null>(null);
   112→  const [editingEntity, setEditingEntity] = useState<VehiculoResponse | null>(null);
   113→  const [formData, setFormData] = useState<FormData>(EMPTY_FORM);
   114→  const [deleteTarget, setDeleteTarget] = useState<VehiculoResponse | null>(null);
   115→
   116→  // Reporte movimiento mensual
   117→  const [reporteVehiculo, setReporteVehiculo] = useState<VehiculoResponse | null>(null);
   118→  const [reporteData, setReporteData] = useState<ReporteMovimientoMensualResponse | null>(null);
   119→  const [reporteLoading, setReporteLoading] = useState(false);
   120→  const [reporteMes, setReporteMes] = useState(new Date().getMonth() + 1);
   121→  const [reporteAnio, setReporteAnio] = useState(new Date().getFullYear());
   122→
   123→  const MESES = [
   124→    { value: 1, label: 'Enero' },
   125→    { value: 2, label: 'Febrero' },
   126→    { value: 3, label: 'Marzo' },
   127→    { value: 4, label: 'Abril' },
   128→    { value: 5, label: 'Mayo' },
   129→    { value: 6, label: 'Junio' },
   130→    { value: 7, label: 'Julio' },
   131→    { value: 8, label: 'Agosto' },
   132→    { value: 9, label: 'Septiembre' },
   133→    { value: 10, label: 'Octubre' },
   134→    { value: 11, label: 'Noviembre' },
   135→    { value: 12, label: 'Diciembre' },
   136→  ];
   137→
   138→  const anios = Array.from({ length: 5 }, (_, i) => new Date().getFullYear() - 2 + i);
   139→
   140→  const handleOpenReporte = (vehiculo: VehiculoResponse) => {
   141→    setReporteVehiculo(vehiculo);
   142→    setReporteData(null);
   143→    setReporteMes(new Date().getMonth() + 1);
   144→    setReporteAnio(new Date().getFullYear());
   145→  };
   146→
   147→  const handleBuscarReporte = async () => {
   148→    if (!reporteVehiculo) return;
   149→    setReporteLoading(true);
   150→    setReporteData(null);
   151→    try {
   152→      const res = await vehiculosApi.reporteMovimientoMensual(reporteVehiculo.id, reporteMes, reporteAnio);
   153→      setReporteData(res.data);
   154→    } catch {
   155→      addToast({ type: 'error', title: 'Error', message: 'No se pudo generar el reporte de movimiento.' });
   156→    } finally {
   157→      setReporteLoading(false);
   158→    }
   159→  };
   160→
   161→  // Show error as toast
   162→  useEffect(() => {
   163→    if (error) {
   164→      addToast({ type: 'error', title: 'Error', message: error });
   165→    }
   166→  }, [error, addToast]);
   167→
   168→  // Fetch static dropdowns (no choferes — those are loaded by empresa)
   169→  const fetchDropdowns = useCallback(async () => {
   170→    try {
   171→      const [empRes, tvRes, marRes, tcRes] = await Promise.all([
   172→        empresasApi.findAll({ page: 0, perPage: 200 }),
   173→        tiposVehiculoApi.findAll({ page: 0, perPage: 200 }),
   174→        marcasApi.findAll({ page: 0, perPage: 200 }),
   175→        tiposCombustibleApi.findAll({ page: 0, perPage: 200 }),
   176→      ]);
   177→      setDropdowns({
   178→        empresas: empRes.data.content.filter((e) => e.activo),
   179→        tiposVehiculo: tvRes.data.content.filter((e) => e.activo),
   180→        marcas: marRes.data.content.filter((e) => e.activo),
   181→        tiposCombustible: tcRes.data.content.filter((e) => e.activo),
   182→      });
   183→    } catch {
   184→      addToast({ type: 'error', title: 'Error', message: 'No se pudieron cargar los datos de los selectores.' });
   185→    }
   186→  }, [addToast]);
   187→
   188→  // Fetch choferes filtered by empresa
   189→  const fetchChoferesByEmpresa = useCallback(async (empresaId: number) => {
   190→    if (!empresaId) {
   191→      setChoferes([]);
   192→      return;
   193→    }
   194→    setLoadingChoferes(true);
   195→    try {
   196→      const res = await choferesApi.findByEmpresaId(empresaId, { page: 0, perPage: 200 });
   197→      setChoferes(res.data.content.filter((c) => c.activo));
   198→    } catch {
   199→      setChoferes([]);
   200→    } finally {
   201→      setLoadingChoferes(false);
   202→    }
   203→  }, []);
   204→
   205→  useEffect(() => {
   206→    fetchDropdowns();
   207→  }, [fetchDropdowns]);
   208→
   209→  // ---- Form helpers ----
   210→
   211→  const handleOpenCreate = () => {
   212→    setEditingEntity(null);
   213→    setFormData({ ...EMPTY_FORM });
   214→    setChoferes([]);
   215→    setShowForm(true);
   216→  };
   217→
   218→  const handleOpenEdit = async (entity: VehiculoResponse) => {
   219→    setEditingEntity(entity);
   220→    // Load choferes for the entity's empresa before setting form data
   221→    await fetchChoferesByEmpresa(entity.empresa.id);
   222→    setFormData({
   223→      empresaId: entity.empresa.id,
   224→      tipoVehiculoId: entity.tipoVehiculo.id,
   225→      marcaId: entity.marca.id,
   226→      choferId: entity.chofer?.id ?? 0,
   227→      tipoCombustibleId: entity.tipoCombustible.id,
   228→      modelo: entity.modelo || '',
   229→      matricula: entity.matricula,
   230→      numeroMotor: entity.numeroMotor,
   231→      odometro: entity.odometro,
   232→      combustible: entity.combustible,
   233→      ultimoMantenimiento: entity.ultimoMantenimiento || '',
   234→      odometroUltimoMantenimiento: entity.odometroUltimoMantenimiento ? String(entity.odometroUltimoMantenimiento) : '',
   235→      indiceConsumo: entity.indiceConsumo ? String(entity.indiceConsumo) : '',
   236→    });
   237→    setShowForm(true);
   238→  };
   239→
   240→  const handleFieldChange = (key: string, value: string | number) => {
   241→    if (key === 'empresaId') {
   242→      // Reset chofer when empresa changes and reload choferes list
   243→      setFormData((prev) => ({ ...prev, empresaId: value as number, choferId: 0 }));
   244→      fetchChoferesByEmpresa(value as number);
   245→    } else {
   246→      setFormData((prev) => ({ ...prev, [key]: value }));
   247→    }
   248→  };
   249→
   250→  const buildRequestPayload = (): VehiculoRequest => ({
   251→    empresaId: formData.empresaId,
   252→    tipoVehiculoId: formData.tipoVehiculoId,
   253→    marcaId: formData.marcaId,
   254→    choferId: formData.choferId || undefined,
   255→    tipoCombustibleId: formData.tipoCombustibleId,
   256→    modelo: formData.modelo || undefined,
   257→    matricula: formData.matricula,
   258→    numeroMotor: formData.numeroMotor,
   259→    odometro: formData.odometro,
   260→    combustible: formData.combustible,
   261→    ultimoMantenimiento: formData.ultimoMantenimiento || undefined,
   262→    odometroUltimoMantenimiento: formData.odometroUltimoMantenimiento ? Number(formData.odometroUltimoMantenimiento) : undefined,
   263→    indiceConsumo: formData.indiceConsumo ? Number(formData.indiceConsumo) : undefined,
   264→  });
   265→
   266→  const handleSubmit = async (e: React.FormEvent) => {
   267→    e.preventDefault();
   268→    try {
   269→      const payload = buildRequestPayload();
   270→      if (editingEntity) {
   271→        await updateItem(editingEntity.id, payload);
   272→        addToast({ type: 'success', title: 'Vehículo actualizado', message: 'El registro se ha actualizado correctamente.' });
   273→      } else {
   274→        await createItem(payload);
   275→        addToast({ type: 'success', title: 'Vehículo creado', message: 'El nuevo registro se ha creado correctamente.' });
   276→      }
   277→      setShowForm(false);
   278→    } catch {
   279→      // error handled by useCrud → toast via useEffect
   280→    }
   281→  };
   282→
   283→  const handleConfirmDelete = async () => {
   284→    if (!deleteTarget) return;
   285→    try {
   286→      await deleteItem(deleteTarget.id);
   287→      addToast({ type: 'success', title: 'Vehículo eliminado', message: 'El registro se ha eliminado correctamente.' });
   288→      setDeleteTarget(null);
   289→    } catch {
   290→      // error handled by useCrud → toast via useEffect
   291→    }
   292→  };
   293→
   294→  // Search
   295→  const filteredData = search
   296→    ? data.filter((item) => {
   297→        const searchStr = `${item.matricula} ${item.numeroMotor} ${item.modelo} ${item.empresa.nombre} ${item.marca.nombre} ${item.tipoVehiculo.nombre} ${item.chofer?.nombre ?? ''} ${item.chofer?.apellidos ?? ''}`.toLowerCase();
   298→        return searchStr.includes(search.toLowerCase());
   299→      })
   300→    : data;
   301→
   302→  // ---- Render helpers ----
   303→
   304→  const renderSelect = (
   305→    id: string,
   306→    label: string,
   307→    value: number,
   308→    onChange: (v: number) => void,
   309→    options: { id: number; label: string }[],
   310→    required = true,
   311→    placeholder = 'Seleccionar...',
   312→  ) => (
   313→    <div>
   314→      <label htmlFor={id} className="block text-sm font-medium text-gray-700 mb-1.5">
   315→        {label}{required && <span className="text-red-500 ml-0.5">*</span>}
   316→      </label>
   317→      <div className="relative">
   318→        <select
   319→          id={id}
   320→          value={value}
   321→          onChange={(e) => onChange(Number(e.target.value))}
   322→          className="input-field appearance-none pr-8"
   323→          required={required}
   324→        >
   325→          <option value={0}>{placeholder}</option>
   326→          {options.map((opt) => (
   327→            <option key={opt.id} value={opt.id}>{opt.label}</option>
   328→          ))}
   329→        </select>
   330→        <ChevronDown className="absolute right-2 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400 pointer-events-none" />
   331→      </div>
   332→    </div>
   333→  );
   334→
   335→  // ---- Render ----
   336→
   337→  return (
   338→    <div>
   339→      <PageHeader title="Vehículos" description="Gestión de los vehículos del sistema">
   340→        <div className="relative w-64">
   341→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   342→          <input
   343→            type="text"
   344→            placeholder="Buscar..."
   345→            value={search}
   346→            onChange={(e) => setSearch(e.target.value)}
   347→            className="input-field pl-9 py-2 text-sm"
   348→          />
   349→        </div>
   350→        <button onClick={handleOpenCreate} className="btn-primary flex items-center gap-2">
   351→          <Plus className="w-4 h-4" />
   352→          Nuevo
   353→        </button>
   354→      </PageHeader>
   355→
   356→      {/* Table */}
   357→      <div className="card !p-0 overflow-hidden">
   358→        <div className="overflow-x-auto">
   359→          <table className="w-full">
   360→            <thead>
   361→              <tr>
   362→                <th className="table-header px-4 py-3">Matrícula</th>
   363→                <th className="table-header px-4 py-3">Empresa</th>
   364→                <th className="table-header px-4 py-3">Tipo</th>
   365→                <th className="table-header px-4 py-3">Marca</th>
   366→                <th className="table-header px-4 py-3">Modelo</th>
   367→                <th className="table-header px-4 py-3">Tipo Combustible</th>
   368→                <th className="table-header px-4 py-3 text-right">Combustible (L)</th>
   369→                <th className="table-header px-4 py-3">Chofer</th>
   370→                <th className="table-header px-4 py-3 text-right">Odómetro</th>
   371→                <th className="table-header px-4 py-3 text-right">Acciones</th>
   372→              </tr>
   373→            </thead>
   374→            <tbody>
   375→              {loading ? (
   376→                <tr>
   377→                  <td colSpan={10} className="px-4 py-12 text-center text-gray-400">
   378→                    <div className="flex items-center justify-center gap-2">
   379→                      <div className="animate-spin rounded-full h-5 w-5 border-b-2 border-primary-600" />
   380→                      Cargando...
   381→                    </div>
   382→                  </td>
   383→                </tr>
   384→              ) : filteredData.length === 0 ? (
   385→                <tr>
   386→                  <td colSpan={10} className="px-4 py-12 text-center text-gray-400">
   387→                    {search ? 'No se encontraron resultados' : 'No hay registros'}
   388→                  </td>
   389→                </tr>
   390→              ) : (
   391→                filteredData.map((item) => (
   392→                  <tr key={item.id} className="hover:bg-gray-50 transition-colors">
   393→                    <td className="px-4 py-3">
   394→                      <span className="table-cell block font-medium text-gray-900">{item.matricula}</span>
   395→                    </td>
   396→                    <td className="px-4 py-3">
   397→                      <span className="table-cell block">{item.empresa.nombre}</span>
   398→                    </td>
   399→                    <td className="px-4 py-3">
   400→                      <span className="table-cell block">{item.tipoVehiculo.nombre}</span>
   401→                    </td>
   402→                    <td className="px-4 py-3">
   403→                      <span className="table-cell block">{item.marca.nombre}</span>
   404→                    </td>
   405→                    <td className="px-4 py-3">
   406→                      <span className="table-cell block">{item.modelo || '—'}</span>
   407→                    </td>
   408→                    <td className="px-4 py-3">
   409→                      <span className="table-cell block">{item.tipoCombustible.denominacion}</span>
   410→                    </td>
   411→                    <td className="px-4 py-3 text-right">
   412→                      <span className="table-cell block">{item.combustible}</span>
   413→                    </td>
   414→                    <td className="px-4 py-3">
   415→                      <span className="table-cell block">
   416→                        {item.chofer
   417→                          ? `${item.chofer.nombre} ${item.chofer.apellidos}`
   418→                          : <span className="text-gray-400">—</span>}
   419→                      </span>
   420→                    </td>
   421→                    <td className="px-4 py-3 text-right">
   422→                      <span className="table-cell block">{item.odometro.toLocaleString()}</span>
   423→                    </td>
   424→                    <td className="px-4 py-3 text-right">
   425→                      <div className="flex items-center justify-end gap-1">
   426→                        <button
   427→                          onClick={() => handleOpenReporte(item)}
   428→                          className="p-1.5 hover:bg-purple-50 rounded-lg text-gray-400 hover:text-purple-600 transition-colors"
   429→                          title="Reporte de Movimiento"
   430→                        >
   431→                          <BarChart3 className="w-4 h-4" />
   432→                        </button>
   433→                        <button
   434→                          onClick={() => setViewEntity(item)}
   435→                          className="p-1.5 hover:bg-green-50 rounded-lg text-gray-400 hover:text-green-600 transition-colors"
   436→                          title="Ver detalles"
   437→                        >
   438→                          <Eye className="w-4 h-4" />
   439→                        </button>
   440→                        <button
   441→                          onClick={() => handleOpenEdit(item)}
   442→                          className="p-1.5 hover:bg-primary-50 rounded-lg text-gray-400 hover:text-primary-600 transition-colors"
   443→                          title="Editar"
   444→                        >
   445→                          <Pencil className="w-4 h-4" />
   446→                        </button>
   447→                        <button
   448→                          onClick={() => setDeleteTarget(item)}
   449→                          className="p-1.5 hover:bg-red-50 rounded-lg text-gray-400 hover:text-red-600 transition-colors"
   450→                          title="Eliminar"
   451→                        >
   452→                          <Trash2 className="w-4 h-4" />
   453→                        </button>
   454→                      </div>
   455→                    </td>
   456→                  </tr>
   457→                ))
   458→              )}
   459→            </tbody>
   460→          </table>
   461→        </div>
   462→
   463→        {!search && (
   464→          <div className="px-4 pb-4">
   465→            <Pagination
   466→              page={page}
   467→              totalPages={totalPages}
   468→              totalElements={totalElements}
   469→              size={size}
   470→              onPageChange={setPage}
   471→            />
   472→          </div>
   473→        )}
   474→      </div>
   475→
   476→      {/* Create / Edit Modal */}
   477→      <Modal
   478→        open={showForm}
   479→        title={editingEntity ? 'Editar Vehículo' : 'Nuevo Vehículo'}
   480→        onClose={() => setShowForm(false)}
   481→        size="xl"
   482→      >
   483→        <form onSubmit={handleSubmit} className="space-y-4">
   484→          {/* Selects de relación */}
   485→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   486→            {renderSelect(
   487→              'empresaId', 'Empresa', formData.empresaId,
   488→              (v) => handleFieldChange('empresaId', v),
   489→              dropdowns.empresas.map((e) => ({ id: e.id, label: e.nombre })),
   490→            )}
   491→            {renderSelect(
   492→              'tipoVehiculoId', 'Tipo de Vehículo', formData.tipoVehiculoId,
   493→              (v) => handleFieldChange('tipoVehiculoId', v),
   494→              dropdowns.tiposVehiculo.map((e) => ({ id: e.id, label: e.nombre })),
   495→            )}
   496→            {renderSelect(
   497→              'marcaId', 'Marca', formData.marcaId,
   498→              (v) => handleFieldChange('marcaId', v),
   499→              dropdowns.marcas.map((e) => ({ id: e.id, label: e.nombre })),
   500→            )}
   501→            {renderSelect(
   502→              'tipoCombustibleId', 'Tipo de Combustible', formData.tipoCombustibleId,
   503→              (v) => handleFieldChange('tipoCombustibleId', v),
   504→              dropdowns.tiposCombustible.map((e) => ({ id: e.id, label: `${e.codigo} - ${e.denominacion}` })),
   505→            )}
   506→            <div>
   507→              <label htmlFor="choferId" className="block text-sm font-medium text-gray-700 mb-1.5">
   508→                Chofer (opcional)
   509→              </label>
   510→              <div className="relative">
   511→                <select
   512→                  id="choferId"
   513→                  value={formData.choferId}
