     1→import { useState, useEffect, useCallback } from 'react';
     2→import { Plus, Pencil, Trash2, Search, ChevronDown, Eye } from 'lucide-react';
     3→import { useCrud } from '@/hooks/useCrud';
     4→import { useToast } from '@/contexts/ToastContext';
     5→import {
     6→  vehiculosApi,
     7→  empresasApi,
     8→  tiposVehiculoApi,
     9→  marcasApi,
    10→  tiposCombustibleApi,
    11→  choferesApi,
    12→} from '@/api/endpoints';
    13→import PageHeader from '@/components/common/PageHeader';
    14→import Pagination from '@/components/common/Pagination';
    15→import Modal from '@/components/ui/Modal';
    16→import ConfirmModal from '@/components/ui/ConfirmModal';
    17→import type {
    18→  VehiculoRequest,
    19→  VehiculoResponse,
    20→  EmpresaResponse,
    21→  TipoVehiculoResponse,
    22→  MarcaResponse,
    23→  TipoCombustibleResponse,
    24→  ChoferResponse,
    25→} from '@/types';
    26→
    27→// ---- Types ----
    28→
    29→interface Dropdowns {
    30→  empresas: EmpresaResponse[];
    31→  tiposVehiculo: TipoVehiculoResponse[];
    32→  marcas: MarcaResponse[];
    33→  tiposCombustible: TipoCombustibleResponse[];
    34→}
    35→
    36→interface FormData {
    37→  empresaId: number;
    38→  tipoVehiculoId: number;
    39→  marcaId: number;
    40→  choferId: number;
    41→  tipoCombustibleId: number;
    42→  modelo: string;
    43→  matricula: string;
    44→  numeroMotor: string;
    45→  odometro: number;
    46→  combustible: number;
    47→  ultimoMantenimiento: string;
    48→  odometroUltimoMantenimiento: string;
    49→  indiceConsumo: string;
    50→}
    51→
    52→const EMPTY_FORM: FormData = {
    53→  empresaId: 0,
    54→  tipoVehiculoId: 0,
    55→  marcaId: 0,
    56→  choferId: 0,
    57→  tipoCombustibleId: 0,
    58→  modelo: '',
    59→  matricula: '',
    60→  numeroMotor: '',
    61→  odometro: 0,
    62→  combustible: 0,
    63→  ultimoMantenimiento: '',
    64→  odometroUltimoMantenimiento: '',
    65→  indiceConsumo: '',
    66→};
    67→
    68→// ---- Componentes auxiliares ----
    69→
    70→function formatDate(dateStr: string): string {
    71→  if (!dateStr) return '—';
    72→  const date = new Date(dateStr + (dateStr.length === 10 ? 'T00:00:00' : ''));
    73→  return date.toLocaleDateString('es-ES', { day: '2-digit', month: '2-digit', year: 'numeric' });
    74→}
    75→
    76→function DetailField({ label, value }: { label: string; value: string }) {
    77→  return (
    78→    <div>
    79→      <dt className="text-xs font-medium text-gray-500">{label}</dt>
    80→      <dd className="mt-0.5 text-sm text-gray-900 font-medium">{value}</dd>
    81→    </div>
    82→  );
    83→}
    84→
    85→// ---- Component ----
    86→
    87→export default function VehiculosPage() {
    88→  const {
    89→    data, loading, saving, totalPages, totalElements, page, size, error,
    90→    setPage, createItem, updateItem, deleteItem,
    91→  } = useCrud<VehiculoRequest, VehiculoResponse>(vehiculosApi);
    92→
    93→  const { addToast } = useToast();
    94→
    95→  // Dropdown data
    96→  const [dropdowns, setDropdowns] = useState<Dropdowns>({
    97→    empresas: [],
    98→    tiposVehiculo: [],
    99→    marcas: [],
   100→    tiposCombustible: [],
   101→  });
   102→
   103→  // Choferes filtrados por empresa
   104→  const [choferes, setChoferes] = useState<ChoferResponse[]>([]);
   105→  const [loadingChoferes, setLoadingChoferes] = useState(false);
   106→
   107→  // UI state
   108→  const [search, setSearch] = useState('');
   109→  const [showForm, setShowForm] = useState(false);
   110→  const [viewEntity, setViewEntity] = useState<VehiculoResponse | null>(null);
   111→  const [editingEntity, setEditingEntity] = useState<VehiculoResponse | null>(null);
   112→  const [formData, setFormData] = useState<FormData>(EMPTY_FORM);
   113→  const [deleteTarget, setDeleteTarget] = useState<VehiculoResponse | null>(null);
   114→
   115→  // Show error as toast
   116→  useEffect(() => {
   117→    if (error) {
   118→      addToast({ type: 'error', title: 'Error', message: error });
   119→    }
   120→  }, [error, addToast]);
   121→
   122→  // Fetch static dropdowns (no choferes — those are loaded by empresa)
   123→  const fetchDropdowns = useCallback(async () => {
   124→    try {
   125→      const [empRes, tvRes, marRes, tcRes] = await Promise.all([
   126→        empresasApi.findAll({ page: 0, perPage: 200 }),
   127→        tiposVehiculoApi.findAll({ page: 0, perPage: 200 }),
   128→        marcasApi.findAll({ page: 0, perPage: 200 }),
   129→        tiposCombustibleApi.findAll({ page: 0, perPage: 200 }),
   130→      ]);
   131→      setDropdowns({
   132→        empresas: empRes.data.content.filter((e) => e.activo),
   133→        tiposVehiculo: tvRes.data.content.filter((e) => e.activo),
   134→        marcas: marRes.data.content.filter((e) => e.activo),
   135→        tiposCombustible: tcRes.data.content.filter((e) => e.activo),
   136→      });
   137→    } catch {
   138→      addToast({ type: 'error', title: 'Error', message: 'No se pudieron cargar los datos de los selectores.' });
   139→    }
   140→  }, [addToast]);
   141→
   142→  // Fetch choferes filtered by empresa
   143→  const fetchChoferesByEmpresa = useCallback(async (empresaId: number) => {
   144→    if (!empresaId) {
   145→      setChoferes([]);
   146→      return;
   147→    }
   148→    setLoadingChoferes(true);
   149→    try {
   150→      const res = await choferesApi.findByEmpresaId(empresaId, { page: 0, perPage: 200 });
   151→      setChoferes(res.data.content.filter((c) => c.activo));
   152→    } catch {
   153→      setChoferes([]);
   154→    } finally {
   155→      setLoadingChoferes(false);
   156→    }
   157→  }, []);
   158→
   159→  useEffect(() => {
   160→    fetchDropdowns();
   161→  }, [fetchDropdowns]);
   162→
   163→  // ---- Form helpers ----
   164→
   165→  const handleOpenCreate = () => {
   166→    setEditingEntity(null);
   167→    setFormData({ ...EMPTY_FORM });
   168→    setChoferes([]);
   169→    setShowForm(true);
   170→  };
   171→
   172→  const handleOpenEdit = async (entity: VehiculoResponse) => {
   173→    setEditingEntity(entity);
   174→    // Load choferes for the entity's empresa before setting form data
   175→    await fetchChoferesByEmpresa(entity.empresa.id);
   176→    setFormData({
   177→      empresaId: entity.empresa.id,
   178→      tipoVehiculoId: entity.tipoVehiculo.id,
   179→      marcaId: entity.marca.id,
   180→      choferId: entity.chofer?.id ?? 0,
   181→      tipoCombustibleId: entity.tipoCombustible.id,
   182→      modelo: entity.modelo || '',
   183→      matricula: entity.matricula,
   184→      numeroMotor: entity.numeroMotor,
   185→      odometro: entity.odometro,
   186→      combustible: entity.combustible,
   187→      ultimoMantenimiento: entity.ultimoMantenimiento || '',
   188→      odometroUltimoMantenimiento: entity.odometroUltimoMantenimiento ? String(entity.odometroUltimoMantenimiento) : '',
   189→      indiceConsumo: entity.indiceConsumo ? String(entity.indiceConsumo) : '',
   190→    });
   191→    setShowForm(true);
   192→  };
   193→
   194→  const handleFieldChange = (key: string, value: string | number) => {
   195→    if (key === 'empresaId') {
   196→      // Reset chofer when empresa changes and reload choferes list
   197→      setFormData((prev) => ({ ...prev, empresaId: value as number, choferId: 0 }));
   198→      fetchChoferesByEmpresa(value as number);
   199→    } else {
   200→      setFormData((prev) => ({ ...prev, [key]: value }));
   201→    }
   202→  };
   203→
   204→  const buildRequestPayload = (): VehiculoRequest => ({
   205→    empresaId: formData.empresaId,
   206→    tipoVehiculoId: formData.tipoVehiculoId,
   207→    marcaId: formData.marcaId,
   208→    choferId: formData.choferId || undefined,
   209→    tipoCombustibleId: formData.tipoCombustibleId,
   210→    modelo: formData.modelo || undefined,
   211→    matricula: formData.matricula,
   212→    numeroMotor: formData.numeroMotor,
   213→    odometro: formData.odometro,
   214→    combustible: formData.combustible,
   215→    ultimoMantenimiento: formData.ultimoMantenimiento || undefined,
   216→    odometroUltimoMantenimiento: formData.odometroUltimoMantenimiento ? Number(formData.odometroUltimoMantenimiento) : undefined,
   217→    indiceConsumo: formData.indiceConsumo ? Number(formData.indiceConsumo) : undefined,
   218→  });
   219→
   220→  const handleSubmit = async (e: React.FormEvent) => {
   221→    e.preventDefault();
   222→    try {
   223→      const payload = buildRequestPayload();
   224→      if (editingEntity) {
   225→        await updateItem(editingEntity.id, payload);
   226→        addToast({ type: 'success', title: 'Vehículo actualizado', message: 'El registro se ha actualizado correctamente.' });
   227→      } else {
   228→        await createItem(payload);
   229→        addToast({ type: 'success', title: 'Vehículo creado', message: 'El nuevo registro se ha creado correctamente.' });
   230→      }
   231→      setShowForm(false);
   232→    } catch {
   233→      // error handled by useCrud → toast via useEffect
   234→    }
   235→  };
   236→
   237→  const handleConfirmDelete = async () => {
   238→    if (!deleteTarget) return;
   239→    try {
   240→      await deleteItem(deleteTarget.id);
   241→      addToast({ type: 'success', title: 'Vehículo eliminado', message: 'El registro se ha eliminado correctamente.' });
   242→      setDeleteTarget(null);
   243→    } catch {
   244→      // error handled by useCrud → toast via useEffect
   245→    }
   246→  };
   247→
   248→  // Search
   249→  const filteredData = search
   250→    ? data.filter((item) => {
   251→        const searchStr = `${item.matricula} ${item.numeroMotor} ${item.modelo} ${item.empresa.nombre} ${item.marca.nombre} ${item.tipoVehiculo.nombre} ${item.chofer?.nombre ?? ''} ${item.chofer?.apellidos ?? ''}`.toLowerCase();
   252→        return searchStr.includes(search.toLowerCase());
   253→      })
   254→    : data;
   255→
   256→  // ---- Render helpers ----
   257→
   258→  const renderSelect = (
   259→    id: string,
   260→    label: string,
   261→    value: number,
   262→    onChange: (v: number) => void,
   263→    options: { id: number; label: string }[],
   264→    required = true,
   265→    placeholder = 'Seleccionar...',
   266→  ) => (
   267→    <div>
   268→      <label htmlFor={id} className="block text-sm font-medium text-gray-700 mb-1.5">
   269→        {label}{required && <span className="text-red-500 ml-0.5">*</span>}
   270→      </label>
   271→      <div className="relative">
   272→        <select
   273→          id={id}
   274→          value={value}
   275→          onChange={(e) => onChange(Number(e.target.value))}
   276→          className="input-field appearance-none pr-8"
   277→          required={required}
   278→        >
   279→          <option value={0}>{placeholder}</option>
   280→          {options.map((opt) => (
   281→            <option key={opt.id} value={opt.id}>{opt.label}</option>
   282→          ))}
   283→        </select>
   284→        <ChevronDown className="absolute right-2 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400 pointer-events-none" />
   285→      </div>
   286→    </div>
   287→  );
   288→
   289→  // ---- Render ----
   290→
   291→  return (
   292→    <div>
   293→      <PageHeader title="Vehículos" description="Gestión de los vehículos del sistema">
   294→        <div className="relative w-64">
   295→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400" />
   296→          <input
   297→            type="text"
   298→            placeholder="Buscar..."
   299→            value={search}
   300→            onChange={(e) => setSearch(e.target.value)}
   301→            className="input-field pl-9 py-2 text-sm"
   302→          />
   303→        </div>
   304→        <button onClick={handleOpenCreate} className="btn-primary flex items-center gap-2">
   305→          <Plus className="w-4 h-4" />
   306→          Nuevo
   307→        </button>
   308→      </PageHeader>
   309→
   310→      {/* Table */}
   311→      <div className="card !p-0 overflow-hidden">
   312→        <div className="overflow-x-auto">
   313→          <table className="w-full">
   314→            <thead>
   315→              <tr>
   316→                <th className="table-header px-4 py-3">Matrícula</th>
   317→                <th className="table-header px-4 py-3">Empresa</th>
   318→                <th className="table-header px-4 py-3">Tipo</th>
   319→                <th className="table-header px-4 py-3">Marca</th>
   320→                <th className="table-header px-4 py-3">Modelo</th>
   321→                <th className="table-header px-4 py-3">Tipo Combustible</th>
   322→                <th className="table-header px-4 py-3 text-right">Combustible (L)</th>
   323→                <th className="table-header px-4 py-3">Chofer</th>
   324→                <th className="table-header px-4 py-3 text-right">Odómetro</th>
   325→                <th className="table-header px-4 py-3 text-right">Acciones</th>
   326→              </tr>
   327→            </thead>
   328→            <tbody>
   329→              {loading ? (
   330→                <tr>
   331→                  <td colSpan={10} className="px-4 py-12 text-center text-gray-400">
   332→                    <div className="flex items-center justify-center gap-2">
   333→                      <div className="animate-spin rounded-full h-5 w-5 border-b-2 border-primary-600" />
   334→                      Cargando...
   335→                    </div>
   336→                  </td>
   337→                </tr>
   338→              ) : filteredData.length === 0 ? (
   339→                <tr>
   340→                  <td colSpan={10} className="px-4 py-12 text-center text-gray-400">
   341→                    {search ? 'No se encontraron resultados' : 'No hay registros'}
   342→                  </td>
   343→                </tr>
   344→              ) : (
   345→                filteredData.map((item) => (
   346→                  <tr key={item.id} className="hover:bg-gray-50 transition-colors">
   347→                    <td className="px-4 py-3">
   348→                      <span className="table-cell block font-medium text-gray-900">{item.matricula}</span>
   349→                    </td>
   350→                    <td className="px-4 py-3">
   351→                      <span className="table-cell block">{item.empresa.nombre}</span>
   352→                    </td>
   353→                    <td className="px-4 py-3">
   354→                      <span className="table-cell block">{item.tipoVehiculo.nombre}</span>
   355→                    </td>
   356→                    <td className="px-4 py-3">
   357→                      <span className="table-cell block">{item.marca.nombre}</span>
   358→                    </td>
   359→                    <td className="px-4 py-3">
   360→                      <span className="table-cell block">{item.modelo || '—'}</span>
   361→                    </td>
   362→                    <td className="px-4 py-3">
   363→                      <span className="table-cell block">{item.tipoCombustible.denominacion}</span>
   364→                    </td>
   365→                    <td className="px-4 py-3 text-right">
   366→                      <span className="table-cell block">{item.combustible}</span>
   367→                    </td>
   368→                    <td className="px-4 py-3">
   369→                      <span className="table-cell block">
   370→                        {item.chofer
   371→                          ? `${item.chofer.nombre} ${item.chofer.apellidos}`
   372→                          : <span className="text-gray-400">—</span>}
   373→                      </span>
   374→                    </td>
   375→                    <td className="px-4 py-3 text-right">
   376→                      <span className="table-cell block">{item.odometro.toLocaleString()}</span>
   377→                    </td>
   378→                    <td className="px-4 py-3 text-right">
   379→                      <div className="flex items-center justify-end gap-1">
   380→                        <button
   381→                          onClick={() => setViewEntity(item)}
   382→                          className="p-1.5 hover:bg-green-50 rounded-lg text-gray-400 hover:text-green-600 transition-colors"
   383→                          title="Ver detalles"
   384→                        >
   385→                          <Eye className="w-4 h-4" />
   386→                        </button>
   387→                        <button
   388→                          onClick={() => handleOpenEdit(item)}
   389→                          className="p-1.5 hover:bg-blue-50 rounded-lg text-gray-400 hover:text-blue-600 transition-colors"
   390→                          title="Editar"
   391→                        >
   392→                          <Pencil className="w-4 h-4" />
   393→                        </button>
   394→                        <button
   395→                          onClick={() => setDeleteTarget(item)}
   396→                          className="p-1.5 hover:bg-red-50 rounded-lg text-gray-400 hover:text-red-600 transition-colors"
   397→                          title="Eliminar"
   398→                        >
   399→                          <Trash2 className="w-4 h-4" />
   400→                        </button>
   401→                      </div>
   402→                    </td>
   403→                  </tr>
   404→                ))
   405→              )}
   406→            </tbody>
   407→          </table>
   408→        </div>
   409→
   410→        {!search && (
   411→          <div className="px-4 pb-4">
   412→            <Pagination
   413→              page={page}
   414→              totalPages={totalPages}
   415→              totalElements={totalElements}
   416→              size={size}
   417→              onPageChange={setPage}
   418→            />
   419→          </div>
   420→        )}
   421→      </div>
   422→
   423→      {/* Create / Edit Modal */}
   424→      <Modal
   425→        open={showForm}
   426→        title={editingEntity ? 'Editar Vehículo' : 'Nuevo Vehículo'}
   427→        onClose={() => setShowForm(false)}
   428→        size="xl"
   429→      >
   430→        <form onSubmit={handleSubmit} className="space-y-4">
   431→          {/* Selects de relación */}
   432→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   433→            {renderSelect(
   434→              'empresaId', 'Empresa', formData.empresaId,
   435→              (v) => handleFieldChange('empresaId', v),
   436→              dropdowns.empresas.map((e) => ({ id: e.id, label: e.nombre })),
   437→            )}
   438→            {renderSelect(
   439→              'tipoVehiculoId', 'Tipo de Vehículo', formData.tipoVehiculoId,
   440→              (v) => handleFieldChange('tipoVehiculoId', v),
   441→              dropdowns.tiposVehiculo.map((e) => ({ id: e.id, label: e.nombre })),
   442→            )}
   443→            {renderSelect(
   444→              'marcaId', 'Marca', formData.marcaId,
   445→              (v) => handleFieldChange('marcaId', v),
   446→              dropdowns.marcas.map((e) => ({ id: e.id, label: e.nombre })),
   447→            )}
   448→            {renderSelect(
   449→              'tipoCombustibleId', 'Tipo de Combustible', formData.tipoCombustibleId,
   450→              (v) => handleFieldChange('tipoCombustibleId', v),
   451→              dropdowns.tiposCombustible.map((e) => ({ id: e.id, label: `${e.codigo} - ${e.denominacion}` })),
   452→            )}
   453→            <div>
   454→              <label htmlFor="choferId" className="block text-sm font-medium text-gray-700 mb-1.5">
   455→                Chofer (opcional)
   456→              </label>
   457→              <div className="relative">
   458→                <select
   459→                  id="choferId"
   460→                  value={formData.choferId}
   461→                  onChange={(e) => handleFieldChange('choferId', Number(e.target.value))}
   462→                  className="input-field appearance-none pr-8"
   463→                  disabled={!formData.empresaId || loadingChoferes}
   464→                >
   465→                  <option value="0">{!formData.empresaId ? 'Seleccione empresa primero' : loadingChoferes ? 'Cargando...' : 'Sin chofer asignado'}</option>
   466→                  {choferes.map((c) => (
   467→                    <option key={c.id} value={c.id}>{`${c.nombre} ${c.apellidos} — ${c.carneIdentidad}`}</option>
   468→                  ))}
   469→                </select>
   470→                <ChevronDown className="absolute right-2 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400 pointer-events-none" />
   471→              </div>
   472→            </div>
   473→          </div>
   474→
   475→          {/* Divider */}
   476→          <div className="border-t border-gray-200" />
   477→
   478→          {/* Datos del vehículo */}
   479→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   480→            <div>
   481→              <label htmlFor="modelo" className="block text-sm font-medium text-gray-700 mb-1.5">
   482→                Modelo
   483→              </label>
   484→              <input
   485→                id="modelo"
   486→                type="text"
   487→                value={formData.modelo}
   488→                onChange={(e) => handleFieldChange('modelo', e.target.value)}
   489→                className="input-field"
   490→                placeholder="Ej: Corolla 2024"
   491→              />
   492→            </div>
   493→            <div>
   494→              <label htmlFor="matricula" className="block text-sm font-medium text-gray-700 mb-1.5">
   495→                Matrícula<span className="text-red-500 ml-0.5">*</span>
   496→              </label>
   497→              <input
   498→                id="matricula"
   499→                type="text"
   500→                value={formData.matricula}
   501→                onChange={(e) => handleFieldChange('matricula', e.target.value)}
   502→                className="input-field"
   503→                placeholder="Ej: A-123-456"
   504→                required
   505→              />
   506→            </div>
   507→            <div>
   508→              <label htmlFor="numeroMotor" className="block text-sm font-medium text-gray-700 mb-1.5">
   509→                No. Motor<span className="text-red-500 ml-0.5">*</span>
   510→              </label>
   511→              <input
   512→                id="numeroMotor"
   513→                type="text"
   514→                value={formData.numeroMotor}
   515→                onChange={(e) => handleFieldChange('numeroMotor', e.target.value)}
   516→                className="input-field"
   517→                placeholder="Ej: M20240001"
   518→                required
   519→              />
   520→            </div>
   521→            <div>
